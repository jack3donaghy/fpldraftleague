<table id="leagueTable">

  <thead>

    <tr>

      <th>Manager</th>

      <th>GW Points</th>

      <th>NewDraft Points (Live)</th>

      <th>PreviousDraft Points</th>

<script>

const csvUrl = "https://docs.google.com/spreadsheets/d/e/2PACX-1vQBEDZp2fIHa82CvjG6sbE0dDlpiZHHQa-5JWo2q3hb-yoBxu0yntj3emupWsEPI7eTNKN66dY5QEA7/pub?gid=0&single=true&output=csv"; // Replace with your published Google Sheet CSV link


function loadTable() {

  fetch(csvUrl)

    .then(response => response.text())

    .then(text => {

      const rows = text.trim().split('\n').map(r => r.split(','));

      const tbody = document.querySelector('#leagueTable tbody');

      tbody.innerHTML = '';


      // Build row objects for sorting

      let dataList = [];

      for (let i = 1; i < rows.length; i++) {

        let row = rows[i];

        dataList.push({

          Manager: row[0],

          Team: row[1],

          GW: Number(row[2]) || 0,

          NewDraft: Number(row[3]) || 0,

          PrevDraft: Number(row[4]) || 0,

          Total: Number(row[5]) || 0,

          Rank: Number(row[6]) || 0,

          Movement: row[7] || '0'

        });

      }


      // Sort by Total Points descending

      dataList.sort((a,b) => b.Total - a.Total);


      // Render rows

      dataList.forEach(d => {

        const tr = document.createElement('tr');

        Object.keys(d).forEach((key, idx) => {

          const td = document.createElement('td');

          td.setAttribute('data-label', key);

          td.textContent = d[key];

          if (key === 'Movement') {

            if (d[key].startsWith('+')) td.className = 'up';

            else if (d[key].startsWith('-')) td.className = 'down';

            else td.className = 'nochange';

          }

          tr.appendChild(td);

        });

        tbody.appendChild(tr);

      });

    })

    .catch(err => console.error('Error loading CSV:', err));

}


// Initial load

loadTable();


// Auto-refresh every 5 minutes (300,000 ms)

setInterval(loadTable, 300000);

</script>
